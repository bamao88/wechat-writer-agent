```python
"""NotebookLM tool wrapper for OpenAI Agent SDK.

Wraps the existing NotebookLM Skill (subprocess-based) in @function_tool decorator
for use with OpenAI Agent SDK.
"""
import asyncio
import os
from typing import Any, Optional
from agents import function_tool, RunContextWrapper


def tool_error_handler(ctx: RunContextWrapper[Any], error: Exception) -> str:
    """Graceful error handling for NotebookLM failures.

    Returns user-friendly message instead of crashing the agent.
    """
    print(f"[NotebookLM Error] {type(error).__name__}: {error}")
    return "Knowledge retrieval temporarily unavailable. I'll answer based on my training knowledge."


@function_tool(failure_error_function=tool_error_handler)
async def query_notebooklm(question: str) -> str:
    """Query NotebookLM knowledge base for relevant information.

    Use this tool when you need to search for specific information from
    the knowledge base to answer questions or gather material for writing.

    Args:
        question: The question or search keywords to find relevant content

    Returns:
        Retrieved content from NotebookLM knowledge base, or error message
    """
    notebook_id = os.getenv("NOTEBOOK_ID")
    if not notebook_id:
        return "Error: NOTEBOOK_ID not configured. Cannot query knowledge base."

    # Path to NotebookLM skill script
    skill_script = os.path.expanduser("~/.claude/skills/notebooklm/notebooklm_tool.py")

    if not os.path.exists(skill_script):
        return f"Error: NotebookLM skill not found at {skill_script}"

    try:
        # Use asyncio subprocess to avoid blocking event loop
        # This is critical for streaming compatibility (see 03-RESEARCH.md pitfall #3)
        process = await asyncio.create_subprocess_exec(
            "python",
            skill_script,
            "--notebook-id", notebook_id,
            "--question", question,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )

        # 30 second timeout for NotebookLM queries
        try:
            stdout, stderr = await asyncio.wait_for(
                process.communicate(),
                timeout=30.0
            )
        except asyncio.TimeoutError:
            process.kill()
            return "Error: NotebookLM query timed out after 30 seconds"

        if process.returncode != 0:
            error_msg = stderr.decode().strip() if stderr else "Unknown error"
            return f"Error: NotebookLM query failed - {error_msg}"

        result = stdout.decode().strip()
        if not result:
            return "No relevant content found in knowledge base."

        return result

    except Exception as e:
        return f"Error: Failed to execute NotebookLM query - {str(e)}"
```

Key design decisions:
- Use `asyncio.create_subprocess_exec` (not subprocess.run) to avoid blocking the async event loop
- 30 second timeout matches existing ERR-02 requirement for subprocess timeout
- Custom error handler for graceful degradation (aligns with future ERR-03 requirement)
- Docstring follows Google format for automatic schema generation
