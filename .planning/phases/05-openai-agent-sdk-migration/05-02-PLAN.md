"""Metrics collection for OpenAI Agent SDK.

Provides OpenAIAgentMetrics dataclass that mirrors AgentRunMetrics from
the Claude Agent SDK, enabling consistent logging and comparison between
the two SDK implementations.
"""
import time
from dataclasses import dataclass, field
from typing import List, Dict, Any, Optional


@dataclass
class OpenAIAgentMetrics:
    """Metrics for OpenAI Agent SDK runs.

    Mirrors the structure of AgentRunMetrics from src/modules/agent_sdk.py
    for compatibility with existing logging infrastructure.
    """

    start_time: float = field(default_factory=time.time)
    end_time: Optional[float] = None
    tool_calls: List[Dict[str, Any]] = field(default_factory=list)
    total_tokens: int = 0
    prompt_tokens: int = 0
    completion_tokens: int = 0
    errors: List[str] = field(default_factory=list)
    system_prompt: Optional[str] = None
    initial_user_message: Optional[str] = None
    messages: List[Dict[str, Any]] = field(default_factory=list)

    # OpenAI SDK specific fields
    agent_name: Optional[str] = None
    model_name: Optional[str] = None

    @property
    def runtime_seconds(self) -> float:
        """Calculate runtime in seconds."""
        if self.end_time is not None:
            return self.end_time - self.start_time
        return time.time() - self.start_time

    @property
    def tool_call_count(self) -> int:
        """Return number of tool calls."""
        return len(self.tool_calls)

    def add_tool_call(
        self,
        tool_name: str,
        tool_input: Dict[str, Any],
        tool_output: str,
        duration_ms: float,
        success: bool = True
    ) -> None:
        """Record a tool call.

        Args:
            tool_name: Name of the tool called
            tool_input: Input arguments to the tool
            tool_output: Output from the tool
            duration_ms: Duration in milliseconds
            success: Whether the call succeeded
        """
        # Extract query params for NotebookLM compatibility
        query_params = tool_input.get('question') or tool_input.get('query') or str(tool_input)

        self.tool_calls.append({
            'tool_name': tool_name,
            'input': tool_input,
            'query_params': query_params,
            'result': tool_output,
            'result_summary': tool_output[:200] if tool_output else None,
            'result_length': len(tool_output) if tool_output else 0,
            'duration_ms': duration_ms,
            'success': success,
            'timestamp': time.time()
        })

    def finalize(self, usage: Any = None) -> None:
        """Finalize metrics after agent run.

        Args:
            usage: Usage object from context_wrapper.usage
        """
        self.end_time = time.time()

        if usage:
            self.total_tokens = getattr(usage, 'total_tokens', 0)
            self.prompt_tokens = getattr(usage, 'input_tokens', 0)
            self.completion_tokens = getattr(usage, 'output_tokens', 0)

    def to_dict(self) -> Dict[str, Any]:
        """Convert metrics to dictionary for logging."""
        return {
            'sdk': 'openai-agents',
            'agent_name': self.agent_name,
            'model_name': self.model_name,
            'runtime_seconds': self.runtime_seconds,
            'tool_call_count': self.tool_call_count,
            'tool_calls': self.tool_calls,
            'total_tokens': self.total_tokens,
            'prompt_tokens': self.prompt_tokens,
            'completion_tokens': self.completion_tokens,
            'errors': self.errors
        }
```

"""OpenAI Agent SDK runner with LiteLLM MiniMax integration.

This module provides an alternative execution path to the Claude Agent SDK
(src/modules/agent_sdk.py), enabling A/B comparison without modifying the
baseline implementation.

Key differences from Claude Agent SDK:
- Uses @function_tool decorator instead of Skills
- Uses LitellmModel for MiniMax API integration
- Uses Runner.run_streamed() instead of query()
- Uses context_wrapper.usage for metrics instead of hooks

Reference: .planning/phases/05-openai-agent-sdk-migration/03-RESEARCH.md
"""
import os
import time
from typing import List, Any, Optional, AsyncIterator

from agents import Agent, Runner, ModelSettings
from litellm import LitellmModel

from ..tools.notebooklm_tool import query_notebooklm
from ..hooks.openai_metrics import OpenAIAgentMetrics


class OpenAIAgentRunner:
    """OpenAI Agent SDK runner with MiniMax API via LiteLLM.

    Parallel implementation to AgentSDKRunner for comparison purposes.
    Claude Agent SDK baseline (agent_sdk.py) remains unchanged.
    """

    def __init__(
        self,
        api_key: str,
        model: str = "minimax/MiniMax-M2.1",
        temperature: float = 0.7,
        notebook_id: Optional[str] = None
    ):
        """Initialize OpenAI Agent SDK runner.

        Args:
            api_key: MiniMax API key (or MINIMAX_API_KEY env var)
            model: Model name in LiteLLM format (default: minimax/MiniMax-M2.1)
            temperature: Generation temperature (default: 0.7)
            notebook_id: NotebookLM notebook ID for tool calling
        """
        self.api_key = api_key or os.getenv("MINIMAX_API_KEY")
        self.model = model
        self.temperature = temperature
        self.notebook_id = notebook_id or os.getenv("NOTEBOOK_ID")

        # Create LiteLLM model wrapper
        self._litellm_model = LitellmModel(
            model=self.model,
            api_key=self.api_key
        )

    def _build_user_message(
        self,
        topic: str,
        search_results: List[Any]
    ) -> str:
        """Build user message from topic and search results.

        Mirrors AgentSDKRunner._build_user_message for consistency.
        """
        message = f"选题：{topic}\n\n"

        if search_results:
            message += "已为你检索到以下素材：\n\n"
            for i, result in enumerate(search_results, 1):
                message += f"【素材 {i}】\n{result.content}\n\n"
                if hasattr(result, 'source') and result.source:
                    message += f"来源：{result.source}\n\n"
            message += "请基于以上素材，结合你的理解，撰写一篇高质量的公众号文章。"
        else:
            message += "未检索到相关素材，请基于你的理解撰写文章，必要时可以使用工具追加检索。"

        return message

    def _create_agent(self, system_prompt: str) -> Agent:
        """Create agent with tools and configuration.

        Args:
            system_prompt: System prompt for the agent

        Returns:
            Configured Agent instance
        """
        # Determine tools based on notebook_id availability
        tools = []
        if self.notebook_id:
            tools.append(query_notebooklm)
            print(f"[OpenAI Agent] NotebookLM tool enabled (notebook_id={self.notebook_id[:20]}...)")
        else:
            print("[OpenAI Agent] WARNING: notebook_id not set, NotebookLM tool not available")

        agent = Agent(
            name="ArticleWriter",
            instructions=system_prompt,
            model=self._litellm_model,
            tools=tools
        )

        # Enable token usage tracking for LiteLLM
        # CRITICAL: Without this, context_wrapper.usage will be empty
        # Reference: 03-RESEARCH.md Pitfall #2
        agent.model_settings = ModelSettings(include_usage=True)

        return agent

    async def generate(
        self,
        topic: str,
        search_results: List[Any],
        system_prompt: str,
        max_turns: int = 10
    ) -> tuple[str, OpenAIAgentMetrics]:
        """Generate article using OpenAI Agent SDK.

        Args:
            topic: Article topic
            search_results: Pre-retrieved search results
            system_prompt: System prompt for article generation
            max_turns: Maximum conversation turns (default: 10)

        Returns:
            Tuple of (generated article text, OpenAIAgentMetrics)
        """
        # Initialize metrics
        metrics = OpenAIAgentMetrics()
        metrics.system_prompt = system_prompt

        # Build user message
        user_message = self._build_user_message(topic, search_results)
        metrics.initial_user_message = user_message

        # Create agent
        agent = self._create_agent(system_prompt)
        metrics.agent_name = agent.name
        metrics.model_name = self.model

        try:
            # Run agent with streaming
            result = await Runner.run_streamed(
                agent,
                input=user_message,
                max_turns=max_turns
            )

            # Process streaming events
            tool_start_times = {}  # Track tool call start times

            async for event in result.stream_events():
                event_record = {
                    'timestamp': time.time(),
                    'type': event.type
                }

                if event.type == "raw_response_event":
                    # Streaming chunks - can print for real-time output
                    pass

                elif event.type == "run_item_stream_event":
                    # Completed items (messages, tool calls)
                    item = event.data
                    if hasattr(item, 'type'):
                        event_record['item_type'] = item.type

                        if item.type == "tool_call_item":
                            # Tool call started
                            tool_name = getattr(item, 'name', 'unknown')
                            tool_input = getattr(item, 'arguments', {})
                            tool_start_times[id(item)] = time.time()
                            print(f"[OpenAI Agent] Tool call: {tool_name}")

                        elif item.type == "tool_call_output_item":
                            # Tool call completed
                            tool_output = getattr(item, 'output', '')
                            # Find matching start time
                            duration_ms = 0
                            for key in list(tool_start_times.keys()):
                                duration_ms = (time.time() - tool_start_times.pop(key)) * 1000
                                break

                            # Record tool call in metrics
                            metrics.add_tool_call(
                                tool_name=getattr(item, 'call_id', 'unknown'),
                                tool_input={},  # Not available in output item
                                tool_output=str(tool_output)[:500],
                                duration_ms=duration_ms,
                                success=not str(tool_output).lower().startswith('error')
                            )

                metrics.messages.append(event_record)

            # Get final output
            final_output = result.final_output or ""

            # Finalize metrics with usage data
            usage = result.context_wrapper.usage if hasattr(result, 'context_wrapper') else None
            metrics.finalize(usage)

            print(f"[OpenAI Agent] Generation complete")
            print(f"  Tool calls: {metrics.tool_call_count}")
            print(f"  Runtime: {metrics.runtime_seconds:.2f}s")
            print(f"  Tokens: {metrics.total_tokens}")

            return final_output, metrics

        except Exception as e:
            metrics.errors.append(str(e))
            metrics.finalize()
            print(f"[OpenAI Agent] Error: {e}")
            raise

    async def generate_simple(
        self,
        topic: str,
        search_results: List[Any],
        system_prompt: str
    ) -> tuple[str, OpenAIAgentMetrics]:
        """Simple non-streaming generation for testing.

        Uses Runner.run() instead of run_streamed() for simpler execution.
        """
        metrics = OpenAIAgentMetrics()
        metrics.system_prompt = system_prompt

        user_message = self._build_user_message(topic, search_results)
        metrics.initial_user_message = user_message

        agent = self._create_agent(system_prompt)
        metrics.agent_name = agent.name
        metrics.model_name = self.model

        try:
            result = await Runner.run(agent, input=user_message)

            # Extract output
            final_output = result.final_output or ""

            # Finalize metrics
            usage = result.context_wrapper.usage if hasattr(result, 'context_wrapper') else None
            metrics.finalize(usage)

            return final_output, metrics

        except Exception as e:
            metrics.errors.append(str(e))
            metrics.finalize()
            raise
```
