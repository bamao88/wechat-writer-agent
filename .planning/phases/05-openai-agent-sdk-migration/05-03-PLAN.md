        """Metrics records tool calls."""
        from src.hooks.openai_metrics import OpenAIAgentMetrics

        metrics = OpenAIAgentMetrics()
        metrics.add_tool_call(
            tool_name='query_notebooklm',
            tool_input={'question': 'test query'},

            duration_ms=150.0,
            success=True
        )

        assert metrics.tool_call_count == 1
        assert metrics.tool_calls[0]['tool_name'] == 'query_notebooklm'
        assert metrics.tool_calls[0]['query_params'] == 'test query'
        assert metrics.tool_calls[0]['success'] == True

    def test_metrics_finalize_with_usage(self):
        """Metrics extracts tokens from usage object."""
        from src.hooks.openai_metrics import OpenAIAgentMetrics

        metrics = OpenAIAgentMetrics()

        # Mock usage object
        usage = MagicMock()
        usage.total_tokens = 1000
        usage.input_tokens = 800
        usage.output_tokens = 200

        metrics.finalize(usage)

        assert metrics.total_tokens == 1000
        assert metrics.prompt_tokens == 800
        assert metrics.completion_tokens == 200
        assert metrics.end_time is not None

    def test_metrics_to_dict(self):
        """Metrics can be converted to dict for logging."""
