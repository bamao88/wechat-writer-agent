   ```python
       """Log when a tool call fails"""

           'error_type': error_info.get('error_type', 'unknown'),
           'error_message': str(error_info.get('error', ''))[:500],
           'timestamp': time.time()
       }
       metrics.tool_lifecycle.append(record)
       print(f"[TOOL-LIFECYCLE] Error: {record['tool_name']} - {record['error_type']}")
       return {}
   ```

5. Add get_lifecycle_summary() function:
   ```python
   def get_lifecycle_summary(metrics: Any) -> Dict[str, Any]:
       """Get summary of tool call lifecycle phases"""
       lifecycle = getattr(metrics, 'tool_lifecycle', [])
       phases = {}
       for record in lifecycle:
           phase = record.get('phase', 'unknown')
           phases[phase] = phases.get(phase, 0) + 1
       return {
           'total_events': len(lifecycle),
           'phases': phases,
           'has_errors': phases.get('error', 0) > 0
       }
   ```

   ```python
   def _handle_tool_error(self, error: Exception, tool_name: str, metrics: AgentRunMetrics) -> str:
       """Handle tool error gracefully, returning fallback message"""
       error_info = {
           'tool_name': tool_name,
           'error_type': type(error).__name__,
           'error': error
       }
       # Log to lifecycle (synchronously, can't await in this context)
       record = {
           'phase': 'error',
           'tool_name': tool_name,
           'error_type': type(error).__name__,
           'error_message': str(error)[:500],
           'timestamp': time.time()
       }
       metrics.tool_lifecycle.append(record)
       metrics.errors.append(f"{tool_name}: {str(error)[:200]}")

       # Return fallback message for the model
       return f"[Tool unavailable] {tool_name} encountered an error. Please continue without this tool."
   ```
