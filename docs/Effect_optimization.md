## 📈 评估标准

### 定量指标

| 指标 | 当前基准 | 优化目标 | 测量方式 |
|------|---------|---------|---------|
| 字数 | 2,268 字符 | 3,000-5,000 字符 | 自动统计 |
| 案例数量 | 2-3 个 | 4-6 个 | 人工统计 |
| 二级标题数 | 5 个 | 4-6 个 | 自动统计 |
| 生成时间 | 1 分钟 | <2 分钟 | 系统记录 |
| Token 使用 | 4,035 | <6,000 | API 返回 |

### 定性指标

采用 5 分制评分：

**1. 标题吸引力** (权重 15%)
- 5分：让人忍不住想点击
- 3分：还可以，有一定吸引力
- 1分：平淡无奇

**2. 开头质量** (权重 20%)
- 5分：前 3 句话就抓住读者
- 3分：还算有趣
- 1分：枯燥的背景介绍

**3. 案例丰富度** (权重 25%)
- 5分：4+ 个真实详细案例
- 3分：2-3 个案例
- 1分：没有或很少案例

**4. 方法可操作性** (权重 20%)
- 5分：有清晰步骤，可直接执行
- 3分：有方法但不够具体
- 1分：泛泛而谈

**5. 语言流畅度** (权重 10%)
- 5分：行云流水，一气呵成
- 3分：基本流畅
- 1分：生硬或啰嗦

**6. 结构清晰度** (权重 10%)
- 5分：逻辑清晰，层次分明
- 3分：结构基本合理
- 1分：混乱无序

**总分计算**:
```
总分 = Σ(单项得分 × 权重)
```

**质量等级**:
- 优秀：4.0-5.0
- 良好：3.0-3.9
- 合格：2.0-2.9
- 需改进：<2.0

```python
def build_enhanced_user_message(topic: str, search_results: list) -> str:
    """构建增强版用户消息"""

    message = f"# 选题\n{topic}\n\n"

    if search_results:
        message += "# 检索到的素材\n\n"
        message += "请重点关注以下素材中的：\n"
        message += "- 真实案例和具体场景\n"
        message += "- 可量化的数据和结果\n"
        message += "- 具体的方法和步骤\n\n"

        for i, result in enumerate(search_results, 1):
            content = result.content[:2000]  # 限制长度
            message += f"## 素材 {i}\n\n{content}\n\n"
            if result.source:

        message += "---\n\n"
        message += "请基于以上素材，结合你的理解，创作一篇满足要求的高质量公众号文章。\n"
```

### 附录 C：多角度检索实现

```python
def multi_angle_retrieval(topic: str, notebook_id=None) -> list:
    """多角度检索素材"""

    # 构建多个检索查询
    queries = [
        topic,  # 主查询
        f"{topic} 案例",  # 案例
        f"{topic} 方法",  # 方法论
        f"{topic} 经验",  # 实践经验
        f"{topic} 问题"   # 常见问题
    ]

    all_results = []
    seen_content = set()

    for query in queries:
        try:
            results = retrieval.search(
                query=query,
                notebook_id=notebook_id,
                max_results=3
            )

            # 去重
            for result in results:
                # 简单去重：基于内容前100字符
                content_hash = result.content[:100]
                if content_hash not in seen_content:
                    seen_content.add(content_hash)
                    all_results.append(result)

        except Exception as e:
            print(f"检索失败 {query}: {e}")
            continue

    # 排序和筛选（保留最相关的5条）
    return rank_and_select(all_results, max_count=5)

def rank_and_select(results: list, max_count: int) -> list:
    """排序并选择最相关的结果"""

    # 简单排序：按长度排序（包含更多信息的优先）
    scored_results = []
    for result in results:
        score = len(result.content)
        # 如果有来源，加分
        if result.source:
            score += 1000
        scored_results.append((score, result))

    # 排序并取前 N 个
    scored_results.sort(reverse=True, key=lambda x: x[0])
    return [r for _, r in scored_results[:max_count]]
```

### 附录 D：素材预处理逻辑

```python
def preprocess_search_results(results: list) -> list:
    """预处理检索结果，提取关键信息"""

    processed = []
    for result in results:
        # 提取关键句（包含数字、案例关键词的句子）
        key_sentences = extract_key_sentences(result.content)

        # 识别案例段落
        case_paragraphs = identify_cases(result.content)

        # 重组素材，突出重点
        if case_paragraphs:
            enhanced_content = f"{key_sentences}\n\n重点案例：\n{case_paragraphs}"
        else:
            enhanced_content = key_sentences

        processed.append(SearchResult(
            content=enhanced_content,
            source=result.source
        ))

    return processed

def extract_key_sentences(text: str) -> str:
    """提取关键句子"""
    # 简单实现：提取包含数字或特定关键词的句子
    import re

    sentences = re.split(r'[。！？\n]', text)
    key_sentences = []

    for sentence in sentences:
        if any(keyword in sentence for keyword in ['案例', '数据', '实践', '方法', '步骤']):
            key_sentences.append(sentence.strip())
        elif re.search(r'\d+', sentence):  # 包含数字
            key_sentences.append(sentence.strip())

    return '。'.join(key_sentences[:10])  # 最多10句

def identify_cases(text: str) -> str:
    """识别案例段落"""
    # 简单实现：查找包含"案例"、"例如"等关键词的段落
    import re

    paragraphs = text.split('\n\n')
    case_paragraphs = []

    for para in paragraphs:
        if any(keyword in para for keyword in ['案例', '例如', '比如', '实践', '经历']):
            case_paragraphs.append(para.strip())

    return '\n\n'.join(case_paragraphs[:3])  # 最多3个案例段落
```

```python
def generate_with_iterative_retrieval(topic, initial_results, api_key):
    """实现手动多轮对话"""

    # 第一轮：生成初稿
    draft = generate_draft(topic, initial_results, api_key)

    # 分析初稿，识别信息缺口
    gaps = analyze_content_gaps(draft)

    if gaps:
        # 第二轮：针对缺口追加检索
        additional_results = []
        for gap in gaps:
            results = retrieval.search(gap['query'])
            additional_results.extend(results)

        # 第三轮：基于补充素材重新生成
        all_results = initial_results + additional_results
        final_article = generate_final(topic, all_results, draft, api_key)

        return final_article

    return draft

def analyze_content_gaps(draft: str) -> list:
    """分析内容缺口"""
    gaps = []

    # 检查案例数量
    case_count = draft.count('案例') + draft.count('例如')
    if case_count < 3:
        gaps.append({'type': 'case', 'query': f'{topic} 案例'})

    # 检查数据
    import re
    numbers = re.findall(r'\d+', draft)
    if len(numbers) < 5:
        gaps.append({'type': 'data', 'query': f'{topic} 数据'})

    # 检查方法论
    if '步骤' not in draft and '方法' not in draft:
        gaps.append({'type': 'method', 'query': f'{topic} 方法步骤'})

    return gaps
```
